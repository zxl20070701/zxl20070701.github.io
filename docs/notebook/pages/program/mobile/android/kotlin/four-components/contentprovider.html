ContentProvider

ContentProvider 用于在不同应用之间共享数据，是 Android 实现跨进程通信的标准方式之一。

核心方法

方法 作用
query 查询数据。
insert 插入数据。
update 更新数据。
delete 删除数据。
getType 返回数据类型（MIME）。

自定义 ContentProvider 示例 

class BookProvider : ContentProvider() {

    override fun onCreate(): Boolean {
        // 初始化数据库等
        return true
    }

    override fun query(
        uri: Uri,
        projection: Array<out String>?,
        selection: String?,
        selectionArgs: Array<out String>?,
        sortOrder: String?
    ): Cursor? {
        return null
    }

    override fun getType(uri: Uri): String? = null

    override fun insert(uri: Uri, values: ContentValues?): Uri? = null

    override fun delete(
        uri: Uri, selection: String?, selectionArgs: Array<out String>?
    ): Int = 0

    override fun update(
        uri: Uri, values: ContentValues?,
        selection: String?, selectionArgs: Array<out String>?
    ): Int = 0
}

<provider

通过 ContentResolver 访问 

// query 返回的 Cursor 用 use 自动关闭
contentResolver.query(
    Uri.parse("content://com.example.provider/book"),
    null, null, null, null
)?.use { cursor ->
    val idIndex = cursor.getColumnIndexOrThrow("_id")
    while (cursor.moveToNext()) {
        val id = cursor.getLong(idIndex)
        // 读取其他列
    }
}

Compose 中查询并展示 

@Composable
fun ContactsScreen() {
    val context = LocalContext.current
    var contacts by remember { mutableStateOf(listOf<String>()) }

    LaunchedEffect(Unit) {
        withContext(Dispatchers.IO) {
            val list = mutableListOf<String>()
            context.contentResolver.query(
                ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
                null, null, null, null
            )?.use { cursor ->
                val nameIndex = cursor.getColumnIndexOrThrow(
                    ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME
                )
                while (cursor.moveToNext()) {
                    list.add(cursor.getString(nameIndex))
                }
            }
            contacts = list
        }
    }

    LazyColumn {
        items(contacts) { name ->
            Text(name, modifier = Modifier.padding(16.dp))
        }
    }
}
