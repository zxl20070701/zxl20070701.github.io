WebView

WebView 用于在 App 内显示网页内容，也可以与 JavaScript 双向交互。 在 Compose 中使用 AndroidView 承载 WebView。

Compose 中嵌入 WebView 

@Composable
fun WebPage(url: String) {
    var webView: WebView? by remember { mutableStateOf(null) }

    AndroidView(
        modifier = Modifier.fillMaxSize(),
        factory = { context ->
            WebView(context).apply {
                settings.apply {
                    javaScriptEnabled = true
                    domStorageEnabled = true
                    setSupportZoom(true)
                }
                webViewClient = WebViewClient()
                webChromeClient = WebChromeClient()
                loadUrl(url)
                webView = this
            }
        }
    )

    // 监听返回键
    BackHandler(enabled = webView?.canGoBack() == true) {
        webView?.goBack()
    }
}

与 JavaScript 交互

Android 调用 JS 

webView?.evaluateJavascript("javascript:showMessage('Hello')", null)

JS 调用 Android 

val jsInterface = object {
    @android.webkit.JavascriptInterface
    fun callFromJs(msg: String) {
        // 在主线程弹 Toast
        Toast.makeText(context, msg, Toast.LENGTH_SHORT).show()
    }
}

webView.addJavascriptInterface(jsInterface, "Android")

加载本地 HTML 

val htmlContent = """
    <html><body>
        <h1>Hello Compose</h1>
        <button onclick="Android.callFromJs('Hello from JS')">点击我</button>
    </body></html>
"""

webView?.loadDataWithBaseURL(null, htmlContent, "text/html", "utf-8", null)

加载进度与错误处理 

@Composable
fun WebPageWithProgress(url: String) {
    var progress by remember { mutableIntStateOf(0) }
    var isError by remember { mutableStateOf(false) }
    var errorMsg by remember { mutableStateOf("") }

    Box(modifier = Modifier.fillMaxSize()) {
        AndroidView(
            modifier = Modifier.fillMaxSize(),
            factory = { context ->
                WebView(context).apply {
                    settings.javaScriptEnabled = true
                    webViewClient = object : WebViewClient() {
                        override fun onPageFinished(view: WebView?, url: String?) {
                            progress = 100
                        }
                        override fun onReceivedError(
                            view: WebView?,
                            request: WebResourceRequest?,
                            error: WebResourceError?
                        ) {
                            isError = true
                            errorMsg = error?.description?.toString() ?: "未知错误"
                        }
                    }
                    webChromeClient = object : WebChromeClient() {
                        override fun onProgressChanged(view: WebView?, newProgress: Int) {
                            progress = newProgress
                        }
                    }
                    loadUrl(url)
                }
            }
        )

        // 进度条
        if (progress < 100) {
            LinearProgressIndicator(
                progress = { progress / 100f },
                modifier = Modifier.fillMaxWidth(),
            )
        }

        // 错误提示
        if (isError) {
            Text(
                text = "加载失败: $errorMsg",
                modifier = Modifier.align(Alignment.Center),
                color = Color.Red
            )
        }
    }
}

常用设置

设置 说明
javaScriptEnabled = true 启用 JavaScript。
domStorageEnabled = true 启用 DOM localStorage / sessionStorage。
setSupportZoom(true) 支持双指缩放。
cacheMode NETWORK_DEFAULT / LOAD_DEFAULT / LOAD_NO_CACHE / LOAD_CACHE_ELSE_NETWORK。
