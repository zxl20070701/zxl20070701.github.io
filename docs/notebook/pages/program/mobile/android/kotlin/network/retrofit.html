Retrofit 网络请求

Retrofit 是 Square 开源的类型安全 HTTP 客户端， 配合 Kotlin 协程与 Moshi / Gson 解析，是事实上的行业标准。

添加依赖

三步走

1. 定义数据类 

data class User(
    val id: Int,
    val name: String,
    val avatar: String
)

2. 定义接口 

interface ApiService {

    @GET("users/{id}")
    suspend fun getUser(@Path("id") id: Int): User

    @GET("users")
    suspend fun listUsers(
        @Query("page") page: Int,
        @Query("size") size: Int = 20
    ): List<User>

    @POST("users")
    suspend fun createUser(@Body user: User): User

    @Multipart
    @POST("upload")
    suspend fun upload(
        @Part file: MultipartBody.Part
    ): UploadResponse

    @GET("download")
    @Streaming // 大文件流式下载
    suspend fun download(): ResponseBody
}

3. 构建 Retrofit 

object ApiClient {
    private val okHttp = OkHttpClient.Builder()
        .connectTimeout(10, TimeUnit.SECONDS)
        .readTimeout(15, TimeUnit.SECONDS)
        .addInterceptor(HttpLoggingInterceptor().apply {
            level = if (BuildConfig.DEBUG) {
                HttpLoggingInterceptor.Level.BODY
            } else {
                HttpLoggingInterceptor.Level.NONE
            }
        })
        // 通用拦截器：加 token、加 UA
        .addInterceptor { chain ->
            val newReq = chain.request().newBuilder()
                .addHeader("Authorization", "Bearer ${TokenHolder.token}")
                .build()
            chain.proceed(newReq)
        }
        .build()

    val api: ApiService = Retrofit.Builder()
        .baseUrl("https://api.example.com/")
        .client(okHttp)
        .addConverterFactory(GsonConverterFactory.create())
        .build()
        .create(ApiService::class.java)
}

在 ViewModel 中调用 

class UserViewModel : ViewModel() {

    private val _user = MutableStateFlow<User?>(null)
    val user = _user.asStateFlow()

    fun load(id: Int) {
        viewModelScope.launch {
            try {
                _user.value = ApiClient.api.getUser(id)
            } catch (e: HttpException) {
                // HTTP 4xx / 5xx
            } catch (e: IOException) {
                // 网络断开 / 超时
            }
        }
    }
}

统一封装网络层

通用返回结构 

data class ApiResponse<T>(
    val code: Int,
    val message: String,
    val data: T?
)

// 密封类表示结果
sealed class Result<out T> {
    data class Success<T>(val data: T) : Result<T>()
    data class Error(val code: Int, val msg: String) : Result<Nothing>()
    object Loading : Result<Nothing>()
}

// 统一入口
suspend fun <T> safeCall(block: suspend () -> ApiResponse<T>): Result<T> {
    return try {
        val resp = block()
        if (resp.code == 0 && resp.data != null) {
            Result.Success(resp.data)
        } else {
            Result.Error(resp.code, resp.message)
        }
    } catch (e: HttpException) {
        Result.Error(e.code(), e.message())
    } catch (e: IOException) {
        Result.Error(-1, "网络异常")
    }
}

// 使用
when (val r = safeCall { ApiClient.api.getUser(1) }) {
    is Result.Success -> show(r.data)
    is Result.Error -> toast(r.msg)
    Result.Loading -> showLoading()
}

文件上传 

suspend fun uploadAvatar(file: File): UploadResponse {
    val reqBody = file.asRequestBody("image/*".toMediaTypeOrNull())
    val part = MultipartBody.Part.createFormData("file", file.name, reqBody)
    return ApiClient.api.upload(part)
}

与 Flow 配合（轮询 / 流式更新） 

fun pollUser(id: Int): Flow<User> = flow {
    while (true) {
        emit(ApiClient.api.getUser(id))
        delay(5000) // 5 秒轮询一次
    }
}.flowOn(Dispatchers.IO)
suspend 函数 + Retrofit 是首选写法； 不要再返回 Call<T> 并调用 execute()（会阻塞线程） 或 enqueue()（回调地狱）。
