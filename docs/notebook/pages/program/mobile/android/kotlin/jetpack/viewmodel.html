ViewModel + StateFlow

ViewModel 负责管理 UI 相关数据，旋转屏幕时数据不丢失； StateFlow 作为状态容器，驱动 Compose 自动重组。 二者搭配构成 Jetpack 官方推荐的 MVVM 架构。

添加依赖

定义 ViewModel 

data class HomeUiState(
    val loading: Boolean = false,
    val list: List<Item> = emptyList(),
    val error: String? = null
)

class HomeViewModel(
    private val repo: HomeRepository = HomeRepository()
) : ViewModel() {

    // 私有可变 + 公有只读，是官方推荐写法
    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    init {
        load()
    }

    fun load() {
        viewModelScope.launch {
            _uiState.update { it.copy(loading = true, error = null) }
            try {
                val data = repo.fetch()
                _uiState.update { it.copy(loading = false, list = data) }
            } catch (e: Exception) {
                _uiState.update { it.copy(loading = false, error = e.message) }
            }
        }
    }
}
使用 MutableStateFlow.update { it.copy(...) } 而不是 value = ...，可在多协程并发修改时保证原子性。

Compose 中订阅 

@Composable
fun HomeScreen(viewModel: HomeViewModel = viewModel()) {
    // 感知生命周期，后台自动停止收集
    val state by viewModel.uiState.collectAsStateWithLifecycle()

    Box(Modifier.fillMaxSize()) {
        when {
            state.loading -> CircularProgressIndicator(Modifier.align(Alignment.Center))
            state.error != null -> Text("出错：${state.error}")
            else -> LazyColumn {
                items(state.list) { ItemRow(it) }
            }
        }
    }
}

一次性事件：SharedFlow

StateFlow 会一直保留最新值，旋转屏幕后会重放，不适合 toast、导航等一次性事件。 使用 SharedFlow： 

class HomeViewModel : ViewModel() {

    sealed class UiEvent {
        data class ShowToast(val msg: String) : UiEvent()
        data class Navigate(val route: String) : UiEvent()
    }

    private val _event = MutableSharedFlow<UiEvent>()
    val event = _event.asSharedFlow()

    fun onRefresh() {
        viewModelScope.launch {
            // ...刷新逻辑
            _event.emit(UiEvent.ShowToast("刷新成功"))
        }
    }
}

// Compose 中监听
@Composable
fun HomeScreen(viewModel: HomeViewModel = viewModel(), nav: NavController) {
    LaunchedEffect(Unit) {
        viewModel.event.collect { e ->
            when (e) {
                is UiEvent.ShowToast -> Toast.makeText(ctx, e.msg, Toast.LENGTH_SHORT).show()
                is UiEvent.Navigate -> nav.navigate(e.route)
            }
        }
    }
}

Repository 数据流：Flow → StateFlow 

class UserViewModel(repo: UserRepository) : ViewModel() {

    // 把 Repository 暴露的 Flow 转换为 StateFlow
    val users: StateFlow<List<User>> = repo.observeUsers()
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = emptyList()
        )
}

带参数创建 ViewModel 

// 1. 定义 Factory
class DetailViewModel(private val id: Int) : ViewModel() {
    companion object {
        fun factory(id: Int) = viewModelFactory {
            initializer { DetailViewModel(id) }
        }
    }
}

// 2. Compose 中调用
val vm: DetailViewModel = viewModel(factory = DetailViewModel.factory(id))
