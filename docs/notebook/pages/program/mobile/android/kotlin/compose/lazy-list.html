LazyColumn / LazyRow 列表

Compose 中用 LazyColumn（纵向）、 LazyRow（横向）、 LazyVerticalGrid（网格）渲染长列表， 仅绘制可见项，等效于 View 时代的 RecyclerView。

基础用法 

@Composable
fun MessageList(messages: List<String>) {
    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        items(messages, key = { it }) { msg ->
            MessageItem(msg)
        }
    }
}

@Composable
fun MessageItem(msg: String) {
    Card(modifier = Modifier.fillMaxWidth()) {
        Text(text = msg, modifier = Modifier.padding(16.dp))
    }
}
传入 key 参数可以让 Compose 在数据集变化时准确定位每一项， 保留局部状态并触发动画，性能更好。

多种 item 类型 

LazyColumn {
    // 单个 header
    item {
        Text("标题", style = MaterialTheme.typography.titleLarge)
    }

    // 普通列表
    items(users) { user ->
        UserRow(user)
    }

    // 索引版本
    itemsIndexed(users) { index, user ->
        Text("#$index ${user.name}")
    }
}

分组粘性头（Sticky Header） 

val grouped = contacts.groupBy { it.firstLetter }

LazyColumn {
    grouped.forEach { (letter, list) ->
        stickyHeader {
            Text(
                text = letter.toString(),
                modifier = Modifier
                    .fillMaxWidth()
                    .background(Color.LightGray)
                    .padding(8.dp)
            )
        }
        items(list) { contact ->
            Text(contact.name)
        }
    }
}

滚动控制与监听 

val listState = rememberLazyListState()

// 1. 点击回到顶部（协程动画）
val scope = rememberCoroutineScope()
FloatingActionButton(onClick = {
    scope.launch { listState.animateScrollToItem(0) }
}) {
    Icon(Icons.Default.ArrowUpward, null)
}

// 2. 是否显示"回到顶部"按钮
val showButton by remember {
    derivedStateOf { listState.firstVisibleItemIndex > 5 }
}

// 3. 加载更多：滚到底部时触发
val shouldLoadMore by remember {
    derivedStateOf {
        val lastVisible = listState.layoutInfo.visibleItemsInfo.lastOrNull()?.index ?: 0
        lastVisible >= listState.layoutInfo.totalItemsCount - 3
    }
}
LaunchedEffect(shouldLoadMore) {
    if (shouldLoadMore) viewModel.loadMore()
}

LazyColumn(state = listState) { /* ... */ }

网格列表 

LazyVerticalGrid(
    columns = GridCells.Adaptive(minSize = 128.dp), // 自适应列数
    // 或 GridCells.Fixed(2)
    contentPadding = PaddingValues(8.dp),
    horizontalArrangement = Arrangement.spacedBy(8.dp),
    verticalArrangement = Arrangement.spacedBy(8.dp)
) {
    items(photos) { photo ->
        AsyncImage(model = photo.url, contentDescription = null)
    }
}

item 动画 

LazyColumn {
    items(messages, key = { it.id }) { msg ->
        MessageItem(
            msg = msg,
            modifier = Modifier.animateItem() // 增删/换位自动加动画
        )
    }
}
