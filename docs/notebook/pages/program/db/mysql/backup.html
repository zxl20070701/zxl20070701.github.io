备份与恢复

使用 mysqldump 备份 

-- 备份整个数据库
mysqldump -u root -p mydb > mydb_backup.sql
-- 备份多个数据库
mysqldump -u root -p --databases db1 db2 > backup.sql
-- 备份所有数据库
mysqldump -u root -p --all-databases > all_backup.sql
-- 备份指定表
mysqldump -u root -p mydb table1 table2 > tables_backup.sql
-- 仅备份结构
mysqldump -u root -p --no-data mydb > schema.sql
-- 仅备份数据
mysqldump -u root -p --no-create-info mydb > data.sql

使用 mysql 恢复 

-- 恢复数据库
mysql -u root -p mydb < mydb_backup.sql
-- 或在 MySQL 命令行内恢复
mysql -u root -p
SOURCE /path/to/mydb_backup.sql;

物理备份 

-- 使用 xtrabackup（Percona 提供）
xtrabackup --backup --target-dir=/backup/mydb
xtrabackup --prepare --target-dir=/backup/mydb
xtrabackup --copy-back --target-dir=/backup/mydb

导出为 CSV 

-- 查询结果导出为文件
SELECT * FROM users INTO OUTFILE '/tmp/users.csv'
FIELDS TERMINATED BY ','
ENCLOSED BY '"'
LINES TERMINATED BY '\n';

-- 从文件导入
LOAD DATA INFILE '/tmp/users.csv'
INTO TABLE users
FIELDS TERMINATED BY ','
ENCLOSED BY '"'
LINES TERMINATED BY '\n';

定时备份脚本示例 

#!/bin/bash
# MySQL 定时备份脚本
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/backup/mysql"
DB_NAME="mydb"

mkdir -p $BACKUP_DIR
mysqldump -u root -p'password' $DB_NAME | gzip > $BACKUP_DIR/${DB_NAME}_${DATE}.sql.gz

# 删除30天前的备份
find $BACKUP_DIR -name "*.sql.gz" -mtime +30 -delete
