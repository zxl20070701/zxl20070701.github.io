聚合管道

概述

MongoDB 的聚合管道(aggregation pipeline)类似于 SQL 的 GROUP BY、JOIN 等操作，通过多个阶段(stage)处理文档。 

db.collection.aggregate([
    {stage1},
    {stage2},
    ...
])

$match 过滤 

// 等价于 WHERE 子句
db.orders.aggregate([
    {$match: {status: "paid"}}
])

$group 分组 

// 等价于 GROUP BY
db.orders.aggregate([
    {
        $group: {
            _id: "$city",
            totalAmount: {$sum: "$amount"},
            count: {$sum: 1}
        }
    }
])

$project 投影 

// 选择、重命名、计算字段
db.users.aggregate([
    {
        $project: {
            name: 1,
            fullName: {$concat: ["$firstName", " ", "$lastName"]},
            _id: 0
        }
    }
])

$sort 排序 

db.orders.aggregate([
    {$sort: {totalAmount: -1}}
])

$limit 和 $skip 

db.orders.aggregate([
    {$limit: 10},
    {$skip: 20}
])

$unwind 展开数组 

// 将数组字段拆分成多条文档
db.users.aggregate([
    {$unwind: "$tags"}
])

$lookup 关联查询 

// 等价于 LEFT JOIN
db.orders.aggregate([
    {
        $lookup: {
            from: "users",
            localField: "userId",
            foreignField: "_id",
            as: "userInfo"
        }
    }
])

$addFields 添加字段 

db.users.aggregate([
    {
        $addFields: {
            ageGroup: {
                $cond: [
                    {$gte: ["$age", 18]},
                    "adult",
                    "minor"
                ]
            }
        }
    }
])

$cond 条件表达式 

db.students.aggregate([
    {
        $project: {
            name: 1,
            grade: {
                $switch: {
                    branches: [
                        {case: {$gte: ["$score", 90]}, then: "A"},
                        {case: {$gte: ["$score", 80]}, then: "B"},
                        {case: {$gte: ["$score", 60]}, then: "C"}
                    ],
                    default: "F"
                }
            }
        }
    }
])

完整示例 

// 统计每个城市的男女比例
db.users.aggregate([
    {$match: {age: {$gte: 18}}},
    {
        $group: {
            _id: "$city",
            total: {$sum: 1},
            males: {$sum: {$cond: [{$eq: ["$gender", "男"]}, 1, 0]}},
            females: {$sum: {$cond: [{$eq: ["$gender", "女"]}, 1, 0]}}
        }
    },
    {$sort: {total: -1}}
])
