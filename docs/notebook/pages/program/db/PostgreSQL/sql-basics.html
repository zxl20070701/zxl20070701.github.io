SQL 基础查询

SELECT 查询 

SELECT * FROM users;
SELECT id, name, age FROM users;
SELECT DISTINCT city FROM users;
SELECT name AS "姓名", age AS "年龄" FROM users;
-- 计算列
SELECT name, age, age + 1 AS "明年年龄" FROM users;

WHERE 条件 

SELECT * FROM users WHERE age > 18;
SELECT * FROM users WHERE name = '张三';
SELECT * FROM users WHERE age BETWEEN 20 AND 30;
SELECT * FROM users WHERE city IN ('北京', '上海', '广州');
SELECT * FROM users WHERE name LIKE '张%';
SELECT * FROM users WHERE email IS NOT NULL;
SELECT * FROM users WHERE age > ALL (SELECT age FROM users WHERE city = '北京');

ORDER BY 

SELECT * FROM users ORDER BY age ASC;
SELECT * FROM users ORDER BY age DESC, name ASC;
SELECT * FROM users ORDER BY 1, 2;

LIMIT 和 OFFSET 

SELECT * FROM users LIMIT 10;
SELECT * FROM users LIMIT 10 OFFSET 20;
-- 分页查询
SELECT * FROM users LIMIT 10 OFFSET (page-1)*10;

INSERT 

INSERT INTO users (name, age, email) VALUES ('张三', 20, 'test@test.com');
INSERT INTO users (name, age) VALUES ('李四', 25), ('王五', 30);
-- 从查询插入
INSERT INTO users_archive SELECT * FROM users WHERE age > 60;

UPDATE 

UPDATE users SET age = 25 WHERE name = '张三';
UPDATE users SET age = age + 1;
UPDATE users SET (age, city) = (25, '北京') WHERE name = '张三';
-- 基于其他表更新
UPDATE orders o SET status = 'confirmed'
FROM users u WHERE o.user_id = u.id AND u.city = '北京';

DELETE 

DELETE FROM users WHERE id = 1;
DELETE FROM users WHERE age < 18;
DELETE FROM users WHERE id IN (SELECT user_id FROM orders);
-- 基于其他表删除
DELETE FROM orders o USING users u
WHERE o.user_id = u.id AND u.city = '北京';

DISTINCT 去重 

SELECT DISTINCT ON (city) city, name, age
FROM users ORDER BY city, age DESC;
